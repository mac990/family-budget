  1. 試算表 → 擴充功能 → Apps Script → 清空後貼上。
  2. 第一行 TOKEN 改成家庭密碼 → 儲存。
  3. 第一次：部署 → 新增部署作業 → 網頁應用程式 → 執行身分「我」、存取權「所有人」。

var TOKEN = '請改成你們家的密碼';
var CHUNK = 45000;

function doGet(e) {
  var p = (e && e.parameter) || {};
  if (p.token !== TOKEN) return out({ error: 'token' });
  return out({ data: readDB() });
}

function doPost(e) {
  var body;
  try { body = JSON.parse(e.postData.contents); } catch (err) { return out({ error: 'json' }); }
  if (body.token !== TOKEN) return out({ error: 'token' });
  if (!body.data) return out({ error: 'nodata' });
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var db = ss.getSheetByName('DB');
    if (db && db.getLastRow() > 0 && !ss.getSheetByName('DB_舊版')) {
      var old = readDB();
      if (!old || old.schema !== 5) db.copyTo(ss).setName('DB_舊版');
    }
    writeDB(body.data);
    writeOverview(body.data);
  } finally { lock.releaseLock(); }
  return out({ ok: true });
}

function out(o) {
  return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON);
}
function sheet(name) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  return ss.getSheetByName(name) || ss.insertSheet(name);
}
function readDB() {
  var sh = sheet('DB');
  var last = sh.getLastRow();
  if (!last) return null;
  var s = sh.getRange(1, 1, last, 1).getValues().map(function (r) { return r[0]; }).join('');
  if (!s) return null;
  try { return JSON.parse(s); } catch (e) { return null; }
}
function writeDB(data) {
  var sh = sheet('DB');
  sh.clearContents();
  var s = JSON.stringify(data), rows = [];
  for (var i = 0; i < s.length; i += CHUNK) rows.push([s.substring(i, i + CHUNK)]);
  if (rows.length) sh.getRange(1, 1, rows.length, 1).setValues(rows);
}
function writeOverview(d) {
  var sh = sheet('總覽');
  sh.clearContents();
  var mem = {};
  (d.members || []).forEach(function (m) { mem[m.id] = m.name; });
  var name = function (id) { return mem[id] || '共同'; };
  var ym = Utilities.formatDate(new Date(), 'Asia/Taipei', 'yyyy-MM');
  var pay = (d.payments || {})[ym] || {};
  var ov = (d.overrides || {})[ym] || {};
  var rows = [['更新時間', new Date(d.updatedAt || Date.now())], [], ['本月 ' + ym + ' 固定項目'],
    ['項目', '金額', '扣款日', '負擔人', '類型', '狀態', '勾選人', '時間']];
  var total = 0, paid = 0;
  (d.items || []).slice().sort(function (a, b) { return (a.day || 99) - (b.day || 99); }).forEach(function (it) {
    var amt = ov[it.id] != null ? Number(ov[it.id]) : Number(it.amount || 0);
    total += amt;
    var p = pay[it.id];
    if (p) paid += amt;
    rows.push([it.name, amt, it.day || '', name(it.memberId), it.type === 'loan' ? '貸款' : '帳單',
      p ? '已繳' : '待繳', p ? name(p.by) : '', p ? new Date(p.at) : '']);
  });
  rows.push(['合計', total, '', '', '', '已繳 ' + paid + ' / 待繳 ' + (total - paid), '', '']);
  rows.push([]); rows.push(['罰單']); rows.push(['名稱', '金額', '到期日', '負擔人', '狀態']);
  (d.fines || []).forEach(function (f) {
    rows.push([f.name, Number(f.amount || 0), f.due || '', name(f.memberId), f.paid ? '已繳' : '未繳']);
  });
  var w = 8;
  rows = rows.map(function (r) { while (r.length < w) r.push(''); return r; });
  sh.getRange(1, 1, rows.length, w).setValues(rows);
}
